async function loadSMCData() { const icon = document.getElementById("refresh-icon"); if (icon) icon.classList.add("animate-spin"); document.getElementById("scan-status").innerText = "正在讀取最新數據..."; try { const res = await fetch(`data.json?_t=${Date.now()}`); if (!res.ok) { throw new Error(`無法取得檔案 (HTTP ${res.status})，請確認 GitHub Actions 是否已運行成功並生成 data.json`); } const text = await res.text(); let data; try { data = JSON.parse(text); } catch (parseErr) { throw new Error("data.json 格式損毀或含有不合規字元，請重新執行 Actions！"); } if (!Array.isArray(data) || data.length === 0) { throw new Error("data.json 內容為空，代表未能成功抓取到任何股票行情。"); } analyzedResults = data; renderResults(); if (analyzedResults.length > 0) { selectStock(analyzedResults[0]); } document.getElementById("scan-status").innerText = "數據已是最新狀態"; } catch (err) { console.error(err); document.getElementById("scan-status").innerText = "讀取失敗"; document.getElementById("stock-table-body").innerHTML = ` ⚠️️ ${err.message} `; } finally { if (icon) icon.classList.remove("animate-spin"); } }
